  try {
    const search = String(req.query.search || "").trim().slice(0, 100);
    const category = String(req.query.category || "").trim().slice(0, 80);

    const page = Math.max(
      1,
      Number.parseInt(req.query.page, 10) || 1
    );

    const limit = Math.min(
      50,
      Math.max(1, Number.parseInt(req.query.limit, 10) || 20)
    );

    const conditions = ["p.is_active = 1"];
    const params = [];

    if (search) {
      conditions.push("(p.name LIKE ? OR p.description LIKE ?)");
      params.push(`%${search}%`, `%${search}%`);
    }

    if (category) {
      conditions.push("c.slug = ?");
      params.push(category.toLowerCase());
    }

    const where = conditions.join(" AND ");

    const countResult = db.prepare(`
      SELECT COUNT(*) AS total
      FROM products p
      LEFT JOIN categories c ON c.id = p.category_id
      WHERE ${where}
    `).get(...params);

    const products = db.prepare(`
      SELECT
        p.id,
        p.name,
        p.slug,
        p.description,
        p.price,
        p.original_price AS originalPrice,
        p.stock,
        p.image_url AS imageUrl,
        p.sku,
        c.name AS category,
        c.slug AS categorySlug
      FROM products p
      LEFT JOIN categories c ON c.id = p.category_id
      WHERE ${where}
      ORDER BY p.created_at DESC, p.id DESC
      LIMIT ? OFFSET ?
    `).all(...params, limit, (page - 1) * limit);

    res.json({
      success: true,
      count: products.length,
      total: countResult.total,
      page,
      limit,
      data: products
    });
  } catch (error) {
    console.error("Products error:", error);

    res.status(500).json({
      success: false,
      message: "Unable to fetch products"
    });
  }
});

// Existing tables: preserved
db.exec(`
  CREATE TABLE IF NOT EXISTS categories (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL UNIQUE,
    slug TEXT NOT NULL UNIQUE,
    description TEXT DEFAULT '',
    created_at TEXT DEFAULT CURRENT_TIMESTAMP
  );

  CREATE TABLE IF NOT EXISTS products (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    slug TEXT NOT NULL UNIQUE,
    description TEXT DEFAULT '',
    category_id INTEGER,
    price REAL NOT NULL CHECK(price >= 0),
    original_price REAL CHECK(original_price IS NULL OR original_price >= 0),
    stock INTEGER NOT NULL DEFAULT 0 CHECK(stock >= 0),
    image_url TEXT DEFAULT '',
    sku TEXT UNIQUE,
    is_active INTEGER NOT NULL DEFAULT 1,
    created_at TEXT DEFAULT CURRENT_TIMESTAMP,
    updated_at TEXT DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(category_id) REFERENCES categories(id)
  );

  CREATE TABLE IF NOT EXISTS inventory_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    product_id INTEGER NOT NULL,
    change_amount INTEGER NOT NULL,
    previous_stock INTEGER NOT NULL,
    new_stock INTEGER NOT NULL,
    reason TEXT DEFAULT 'Manual update',
    created_at TEXT DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(product_id) REFERENCES products(id)
  );

  CREATE INDEX IF NOT EXISTS idx_products_category
  ON products(category_id);

  CREATE INDEX IF NOT EXISTS idx_products_active
  ON products(is_active);

  -- New admin table
  CREATE TABLE IF NOT EXISTS admin_users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    email TEXT NOT NULL UNIQUE COLLATE NOCASE,
    password_hash TEXT NOT NULL,
    role TEXT NOT NULL DEFAULT 'admin',
    is_active INTEGER NOT NULL DEFAULT 1,
    created_at TEXT DEFAULT CURRENT_TIMESTAMP,
    updated_at TEXT DEFAULT CURRENT_TIMESTAMP
  );

  -- Admin activity history
  CREATE TABLE IF NOT EXISTS admin_audit_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    admin_id INTEGER,
    action TEXT NOT NULL,
    entity_type TEXT NOT NULL,
    entity_id INTEGER,
    details TEXT DEFAULT '',
    created_at TEXT DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(admin_id) REFERENCES admin_users(id)
  );
`);

import { Router } from "express";
import db from "../database.js";
import { requireRider } from "../middleware/rider-auth.js";

const router = Router();

router.use(requireRider);

// GET /api/rider/orders - only this rider's assigned orders
router.get("/orders", (req, res) => {
  try {
    const orders = db.prepare(`
      SELECT
        id,
        order_number,
        customer_name,
        customer_phone,
        delivery_address,
        delivery_city,
        delivery_state,
        delivery_pincode,
        total_amount,
        payment_method,
        payment_status,
        status,
        created_at,
        updated_at
      FROM orders
      WHERE assigned_rider_id = ?
      ORDER BY
        CASE WHEN status = 'Out for Delivery' THEN 0 ELSE 1 END,
        created_at DESC,
        id DESC
    `).all(req.rider.id);

    return res.json({
      success: true,
      data: orders
    });
  } catch (error) {
    console.error("Rider orders error:", error);
    return res.status(500).json({
      success: false,
      message: "Unable to load assigned orders"
    });
  }
});

// GET /api/rider/orders/:id - only assigned order details
router.get("/orders/:id", (req, res) => {
  try {
    const id = Number(req.params.id);

    if (!Number.isSafeInteger(id) || id < 1) {
      return res.status(400).json({
        success: false,
        message: "Invalid order ID"
      });
    }

    const order = db.prepare(`
      SELECT
        id,
        order_number,
        customer_name,
        customer_phone,
        delivery_address,
        delivery_city,
        delivery_state,
        delivery_pincode,
        total_amount,
        payment_method,
        payment_status,
        status,
        created_at,
        updated_at
      FROM orders
      WHERE id = ? AND assigned_rider_id = ?
    `).get(id, req.rider.id);

    if (!order) {
      return res.status(404).json({
        success: false,
        message: "Assigned order not found"
      });
    }

    const items = db.prepare(`
      SELECT
        product_name,
        sku,
        quantity,
        unit_price,
        line_total
      FROM order_items
      WHERE order_id = ?
      ORDER BY id
    `).all(id);

    const events = db.prepare(`
      SELECT
        old_status,
        new_status,
        note,
        created_at
      FROM order_events
      WHERE order_id = ?
      ORDER BY created_at DESC, id DESC
    `).all(id);

    return res.json({
      success: true,
      data: {
        ...order,
        items,
        events
      }
    });
  } catch (error) {
    console.error("Rider order detail error:", error);
    return res.status(500).json({
      success: false,
      message: "Unable to load order details"
    });
  }
});

// PATCH /api/rider/orders/:id/delivered
router.patch("/orders/:id/delivered", (req, res) => {
  try {
    const id = Number(req.params.id);

    if (!Number.isSafeInteger(id) || id < 1) {
      return res.status(400).json({
        success: false,
        message: "Invalid order ID"
      });
    }

    const completeDelivery = db.transaction(() => {
      const order = db.prepare(`
        SELECT id, order_number, status
        FROM orders
        WHERE id = ? AND assigned_rider_id = ?
      `).get(id, req.rider.id);

      if (!order) {
        return { error: "NOT_FOUND" };
      }

      if (order.status !== "Out for Delivery") {
        return {
          error: "INVALID_STATUS",
          current: order.status
        };
      }

      const result = db.prepare(`
        UPDATE orders
        SET status = 'Delivered',
            updated_at = CURRENT_TIMESTAMP
        WHERE id = ?
          AND assigned_rider_id = ?
          AND status = 'Out for Delivery'
      `).run(id, req.rider.id);

      if (result.changes !== 1) {
        return { error: "UPDATE_FAILED" };
      }

      db.prepare(`
        INSERT INTO order_events (
          order_id,
          old_status,
          new_status,
          note,
          changed_by
        )
        VALUES (?, ?, ?, ?, NULL)
      `).run(
        id,
        "Out for Delivery",
        "Delivered",
        `Delivered by rider ${req.rider.name}`
      );

      return {
        success: true,
        order_number: order.order_number,
        old_status: "Out for Delivery",
        new_status: "Delivered"
      };
    });

    const result = completeDelivery();

    if (result.error === "NOT_FOUND") {
      return res.status(404).json({
        success: false,
        message: "Assigned order not found"
      });
    }

    if (result.error === "INVALID_STATUS") {
      return res.status(409).json({
        success: false,
        message: `Order cannot be delivered from status ${result.current}`
      });
    }

    if (result.error === "UPDATE_FAILED") {
      return res.status(409).json({
        success: false,
        message: "Order status changed. Please refresh."
      });
    }

    return res.json({
      success: true,
      message: "Order marked as delivered",
      data: result
    });
  } catch (error) {
    console.error("Rider delivery update error:", error);
    return res.status(500).json({
      success: false,
      message: "Unable to complete delivery"
    });
  }
});

export default router;

import { Router } from "express";
import db from "../database.js";
import { requireAdmin } from "../middleware/auth.js";

const router = Router();

router.use(requireAdmin);

// POST /api/admin/orders/:id/assign-rider
router.post("/:id/assign-rider", (req, res) => {
  try {
    const orderId = Number(req.params.id);
    const riderId = Number(req.body?.rider_id);

    if (!Number.isSafeInteger(orderId) || orderId < 1) {
      return res.status(400).json({
        success: false,
        message: "Invalid order ID"
      });
    }

    if (!Number.isSafeInteger(riderId) || riderId < 1) {
      return res.status(400).json({
        success: false,
        message: "Valid rider ID is required"
      });
    }

    const assignOrder = db.transaction(() => {
      const order = db.prepare(`
        SELECT id, order_number, status, assigned_rider_id
        FROM orders
        WHERE id = ?
      `).get(orderId);

      if (!order) {
        return { error: "ORDER_NOT_FOUND" };
      }

      if (order.status !== "Ready for Dispatch") {
        return {
          error: "INVALID_ORDER_STATUS",
          current: order.status
        };
      }

      const rider = db.prepare(`
        SELECT id, name, phone, email
        FROM riders
        WHERE id = ? AND is_active = 1
      `).get(riderId);

      if (!rider) {
        return { error: "RIDER_NOT_FOUND" };
      }

      const update = db.prepare(`
        UPDATE orders
        SET
          assigned_rider_id = ?,
          status = 'Out for Delivery',
          updated_at = CURRENT_TIMESTAMP
        WHERE id = ?
          AND status = 'Ready for Dispatch'
          AND assigned_rider_id IS NULL
      `).run(riderId, orderId);

      if (update.changes !== 1) {
        return { error: "ASSIGNMENT_CONFLICT" };
      }

      db.prepare(`
        INSERT INTO order_events (
          order_id,
          old_status,
          new_status,
          note,
          changed_by
        )
        VALUES (?, ?, ?, ?, ?)
      `).run(
        orderId,
        "Ready for Dispatch",
        "Out for Delivery",
        `Assigned to rider ${rider.name} (${rider.phone})`,
        req.admin.id
      );

      return {
        success: true,
        order_number: order.order_number,
        rider
      };
    });

    const result = assignOrder();

    if (result.error === "ORDER_NOT_FOUND") {
      return res.status(404).json({
        success: false,
        message: "Order not found"
      });
    }

    if (result.error === "RIDER_NOT_FOUND") {
      return res.status(404).json({
        success: false,
        message: "Active rider not found"
      });
    }

    if (result.error === "INVALID_ORDER_STATUS") {
      return res.status(409).json({
        success: false,
        message: `Only Ready for Dispatch orders can be assigned. Current status: ${result.current}`
      });
    }

    if (result.error === "ASSIGNMENT_CONFLICT") {
      return res.status(409).json({
        success: false,
        message: "Order is already assigned or has changed. Refresh and try again."
      });
    }

    return res.json({
      success: true,
      message: "Order assigned and marked Out for Delivery",
      data: result
    });
  } catch (error) {
    console.error("Rider assignment error:", error);

    return res.status(500).json({
      success: false,
      message: "Unable to assign rider"
    });
  }
});

export default router;

function getToken() {
  return localStorage.getItem(TOKEN_KEY)
}

function saveSession(token: string, customer: Customer) {
  localStorage.setItem(TOKEN_KEY, token)
  localStorage.setItem('riya-customer', JSON.stringify(customer))
}

export function getCustomerSession() {
  const token = getToken()
  const rawCustomer = localStorage.getItem('riya-customer')

  if (!token || !rawCustomer) return null

  try {
    return {
      token,
      customer: JSON.parse(rawCustomer) as Customer
    }
  } catch {
    localStorage.removeItem(TOKEN_KEY)
    localStorage.removeItem('riya-customer')
    return null
  }
}

export function clearCustomerSession() {
  localStorage.removeItem(TOKEN_KEY)
  localStorage.removeItem('riya-customer')
}

async function customerRequest<T>(
  endpoint: string,
  options: RequestInit = {},
  authenticated = true
): Promise<T> {
  const token = getToken()

  const response = await fetch(`${API_URL}${endpoint}`, {
    ...options,
    headers: {
      'Content-Type': 'application/json',
      ...(authenticated && token
        ? { Authorization: `Bearer ${token}` }
        : {}),
      ...options.headers
    }
  })

  const result = await response.json().catch(() => null)

  if (!response.ok || result?.success === false) {
    if (response.status === 401 && authenticated) {
      clearCustomerSession()
    }

    throw new Error(
      result?.message || `Request failed (${response.status})`
    )
  }

  return result as T
}

async function request<T>(endpoint: string): Promise<ApiResponse<T>> {
  const response = await fetch(`${API_URL}${endpoint}`)

    throw new Error(`API request failed: ${response.status}`)

export function getProducts(search = '', category = '') {
  const params = new URLSearchParams()

  if (search.trim()) params.set('search', search.trim())
  if (category && category !== 'All') {
    params.set('category', category.toLowerCase().replaceAll(' ', '-').replaceAll('&-', ''))
  }

  const query = params.toString()

  return request<ApiProduct[]>(`/products${query ? `?${query}` : ''}`)
}

function readToken() {
  return localStorage.getItem(TOKEN_KEY) || "";
}
function getMessage(data: AnyRecord, fallback: string) {
  return data?.message || data?.error || fallback;
}
function getList(data: AnyRecord): AnyRecord[] {
  if (Array.isArray(data)) return data;
  if (Array.isArray(data?.data)) return data.data;
  if (Array.isArray(data?.orders)) return data.orders;
  if (Array.isArray(data?.data?.orders)) return data.data.orders;
  return [];
}
function money(value: any) {
  const number = Number(value || 0);
  return number.toLocaleString("en-IN", { style: "currency", currency: "INR", maximumFractionDigits: 0 });
}
function orderId(order: AnyRecord) {
  return order?.id ?? order?.order_id ?? order?._id ?? "—";
}
function customerName(order: AnyRecord) {
  return order?.customer_name || order?.customer?.name || order?.shipping_name || order?.name || "Customer";
}
function customerPhone(order: AnyRecord) {
  return order?.customer_phone || order?.customer?.phone || order?.phone || order?.shipping_phone || "";
}
function addressOf(order: AnyRecord) {
  const a = order?.shipping_address || order?.delivery_address || order?.address || order?.customer?.address;
  if (typeof a === "string") return a;
  if (a && typeof a === "object") {
    return [a.address_line1, a.address_line2, a.address, a.landmark, a.city, a.state, a.pincode, a.postal_code]
      .filter(Boolean).join(", ");
  }
  return [order?.address_line1, order?.city, order?.state, order?.pincode].filter(Boolean).join(", ");
}
function statusOf(order: AnyRecord) {
  return order?.status || order?.order_status || "Assigned";
}
function amountOf(order: AnyRecord) {
  return order?.total_amount ?? order?.total ?? order?.grand_total ?? order?.amount ?? 0;
}
function itemsOf(order: AnyRecord): AnyRecord[] {
  return Array.isArray(order?.items) ? order.items : Array.isArray(order?.order_items) ? order.order_items : [];
}

async function request(path: string, options: RequestInit = {}) {
  const token = readToken();
  const response = await fetch(`${API}${path}`, {
    ...options,
    headers: {
      "Content-Type": "application/json",
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...(options.headers || {})
    }
  });
  const data = await response.json().catch(() => ({}));
  if (!response.ok) {
    if (response.status === 401) {
      localStorage.removeItem(TOKEN_KEY);
      localStorage.removeItem(PROFILE_KEY);
      window.dispatchEvent(new Event("rider-session-expired"));
    }
    throw new Error(getMessage(data, `Request failed (${response.status})`));
  }
  return data;
}

async function apiRequest<T>(
  endpoint: string,
  token?: string,
  options: RequestInit = {},
): Promise<T> {
  const headers = new Headers(options.headers);

  // FormData uploads need the browser-generated multipart boundary.
  // All existing JSON requests continue using application/json.
  if (
    options.body &&
    !(options.body instanceof FormData) &&
    !headers.has('Content-Type')
  ) {
    headers.set('Content-Type', 'application/json');
  }

  if (token) {
    headers.set('Authorization', `Bearer ${token}`);
  }

  const response = await fetch(`${API_URL}${endpoint}`, {
    ...options,
    headers,
  });

  const result = await response.json().catch(() => ({}));

  if (!response.ok) {
    throw new Error(
      result.message || result.error || `Request failed (${response.status})`,
    );
  }

  return result as T;
}

async function saveProduct(event: React.FormEvent<HTMLFormElement>) {
    event.preventDefault();
    clearMessages();

    if (!productForm.name.trim()) {
      setError('Product name is required.');
      return;
    }

    if (!productForm.categoryId) {
      setError('Please select a category.');
      return;
    }

    const price = Number(productForm.price);
    const originalPrice =
      productForm.originalPrice.trim() === ''
        ? null
        : Number(productForm.originalPrice);
    const stock = Number(productForm.stock);

    if (!Number.isFinite(price) || price < 0) {
      setError('Enter a valid product price.');
      return;
    }

    if (
      originalPrice !== null &&
      (!Number.isFinite(originalPrice) || originalPrice < price)
    ) {
      setError('Original price must be greater than or equal to selling price.');
      return;
    }

    if (!Number.isInteger(stock) || stock < 0) {
      setError('Stock must be a whole number greater than or equal to zero.');
      return;
    }

    const payload = {
      name: productForm.name.trim(),
      description: productForm.description.trim(),
      categoryId: Number(productForm.categoryId),
      price,
      originalPrice,
      stock,
      sku: productForm.sku.trim(),
      imageUrl: productForm.imageUrl.trim(),
    };

    setSavingProduct(true);

    try {
      if (editingProduct) {
        await apiRequest(
          `/admin/products/${editingProduct.id}`,
          token,
          {
            method: 'PUT',
            body: JSON.stringify(payload),
          },
        );

        setSuccess('Product updated successfully.');
      } else {
        await apiRequest('/admin/products', token, {
          method: 'POST',
          body: JSON.stringify(payload),
        });

        setSuccess('Product created successfully.');
      }

      setProductModalOpen(false);
      await Promise.all([loadProducts(), loadDashboard()]);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Unable to save product.');
    } finally {
      setSavingProduct(false);
    }
  }

// GET /api/admin/riders
router.get("/", (req, res) => {
  try {
    const riders = db.prepare(`
      SELECT
        r.id,
        r.name,
        r.phone,
        r.email,
        r.is_active,
        r.created_at,
        r.updated_at,
        COUNT(o.id) AS assigned_orders,
        SUM(CASE WHEN o.status = 'Out for Delivery' THEN 1 ELSE 0 END)
          AS active_deliveries
      FROM riders r
      LEFT JOIN orders o ON o.assigned_rider_id = r.id
      GROUP BY r.id
      ORDER BY r.created_at DESC, r.id DESC
    `).all();

    return res.json({
      success: true,
      data: riders
    });
  } catch (error) {
    console.error("Admin riders list error:", error);
    return res.status(500).json({
      success: false,
      message: "Unable to load riders"
    });
  }
});

// POST /api/admin/riders
router.post("/", async (req, res) => {
  try {
    const name = String(req.body?.name || "").trim();
    const phone = String(req.body?.phone || "").trim();
    const email = String(req.body?.email || "").trim().toLowerCase();
    const password = String(req.body?.password || "");

    if (!name || !phone || !password) {
      return res.status(400).json({
        success: false,
        message: "Name, phone and password are required"
      });
    }

    if (name.length > 100 || phone.length > 20 || email.length > 254) {
      return res.status(400).json({
        success: false,
        message: "One or more fields exceed the allowed length"
      });
    }

    if (password.length < 8 || Buffer.byteLength(password, "utf8") > 72) {
      return res.status(400).json({
        success: false,
        message: "Password must be at least 8 characters and at most 72 bytes"
      });
    }

    if (email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      return res.status(400).json({
        success: false,
        message: "Enter a valid email address"
      });
    }

    const passwordHash = await bcrypt.hash(password, 12);

    const result = db.prepare(`
      INSERT INTO riders (
        name,
        phone,
        email,
        password_hash,
        is_active
      )
      VALUES (?, ?, ?, ?, 1)
    `).run(name, phone, email || null, passwordHash);

    const rider = db.prepare(`
      SELECT id, name, phone, email, is_active, created_at
      FROM riders
      WHERE id = ?
    `).get(result.lastInsertRowid);

    return res.status(201).json({
      success: true,
      message: "Rider created successfully",
      data: rider
    });
  } catch (error) {
    if (error.code === "SQLITE_CONSTRAINT_UNIQUE") {
      return res.status(409).json({
        success: false,
        message: "Phone number or email is already registered"
      });
    }

    console.error("Admin create rider error:", error);
    return res.status(500).json({
      success: false,
      message: "Unable to create rider"
    });
  }
});

router.patch("/:id", async (req, res) => {
  try {
    const id = Number(req.params.id);

    if (!Number.isSafeInteger(id) || id < 1) {
      return res.status(400).json({
        success: false,
        message: "Invalid rider ID"
      });
    }

    const existing = db.prepare(`
      SELECT id, name, phone, email, is_active
      FROM riders
      WHERE id = ?
    `).get(id);

    if (!existing) {
      return res.status(404).json({
        success: false,
        message: "Rider not found"
      });
    }

    const name = req.body?.name === undefined
      ? existing.name
      : String(req.body.name).trim();

    const phone = req.body?.phone === undefined
      ? existing.phone
      : String(req.body.phone).trim();

    const email = req.body?.email === undefined
      ? existing.email
      : String(req.body.email).trim().toLowerCase();

    const password = req.body?.password === undefined
      ? ""
      : String(req.body.password);

    const isActive = req.body?.is_active === undefined
      ? existing.is_active
      : req.body.is_active;

    if (
      !name || !phone ||
      name.length > 100 ||
      phone.length > 20 ||
      String(email || "").length > 254
    ) {
      return res.status(400).json({
        success: false,
        message: "Invalid rider name, phone or email"
      });
    }

    if (
      isActive !== 0 &&
      isActive !== 1 &&
      isActive !== false &&
      isActive !== true
    ) {
      return res.status(400).json({
        success: false,
        message: "is_active must be true or false"
      });
    }

    if (email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      return res.status(400).json({
        success: false,
        message: "Enter a valid email address"
      });
    }

    if (
      password &&
      (password.length < 8 || Buffer.byteLength(password, "utf8") > 72)
    ) {
      return res.status(400).json({
        success: false,
        message: "New password must be at least 8 characters and at most 72 bytes"
      });
    }

    const passwordHash = password
      ? await bcrypt.hash(password, 12)
      : null;

    db.prepare(`
      UPDATE riders
      SET
        name = ?,
        phone = ?,
        email = ?,
        password_hash = COALESCE(?, password_hash),
        is_active = ?,
        updated_at = CURRENT_TIMESTAMP
      WHERE id = ?
    `).run(
      name,
      phone,
      email || null,
      passwordHash,
      isActive ? 1 : 0,
      id
    );

    const updated = db.prepare(`
      SELECT id, name, phone, email, is_active, created_at, updated_at
      FROM riders
      WHERE id = ?
    `).get(id);

    return res.json({
      success: true,
      message: "Rider updated successfully",
      data: updated
    });
  } catch (error) {
    if (error.code === "SQLITE_CONSTRAINT_UNIQUE") {
      return res.status(409).json({
        success: false,
        message: "Phone number or email is already registered"
      });
    }

    console.error("Admin update rider error:", error);
    return res.status(500).json({
      success: false,
      message: "Unable to update rider"
    });
  }
});

import { randomUUID } from "node:crypto";

import db from "../database.js";
import { requireCustomer } from "../middleware/customer-auth.js";

const router = Router();

router.use(requireCustomer);

function clean(value) {
  return typeof value === "string" ? value.trim() : "";
}

// Place a Cash on Delivery order
router.post("/", (req, res) => {
  try {
    const {
      items,
      delivery_address,
      delivery_city,
      delivery_state,
      delivery_pincode,
      customer_notes
    } = req.body || {};

    if (!Array.isArray(items) || items.length === 0 || items.length > 50) {
      return res.status(400).json({
        success: false,
        message: "Please provide between 1 and 50 order items."
      });
    }

    if (
      !clean(delivery_address) ||
      !clean(delivery_city) ||
      !clean(delivery_state) ||
      !/^\d{6}$/.test(clean(delivery_pincode))
    ) {
      return res.status(400).json({
        success: false,
        message: "Valid delivery address, city, state and 6-digit pincode are required."
      });
    }

    const paymentMethod = clean(req.body.payment_method || "COD").toUpperCase();

    if (paymentMethod !== "COD") {
      return res.status(400).json({
        success: false,
        message: "Online payment is not available yet. Please select COD."
      });
    }

    const quantities = new Map();

    for (const item of items) {
      const productId = Number(item.product_id);
      const quantity = Number(item.quantity);

      if (
        !Number.isSafeInteger(productId) ||
        productId <= 0 ||
        !Number.isSafeInteger(quantity) ||
        quantity <= 0 ||
        quantity > 100
      ) {
        return res.status(400).json({
          success: false,
          message: "Invalid product ID or quantity."
        });
      }

      const nextQuantity = (quantities.get(productId) || 0) + quantity;

      if (nextQuantity > 100) {
        return res.status(400).json({
          success: false,
          message: "Maximum quantity per product is 100."
        });
      }

      quantities.set(productId, nextQuantity);
    }

    const deliveryFee = Number(process.env.DELIVERY_FEE || 0);

    if (!Number.isFinite(deliveryFee) || deliveryFee < 0) {
      return res.status(500).json({
        success: false,
        message: "Invalid delivery fee configuration."
      });
    }

    const placeOrder = db.transaction(() => {
      const customer = db.prepare(`
        SELECT id, name, email, phone
        FROM customers
        WHERE id = ? AND is_active = 1
      `).get(req.customer.id);

      if (!customer) {
        const error = new Error("Customer account is not active.");
        error.status = 401;
        throw error;
      }

      const orderItems = [];
      let subtotal = 0;

      for (const [productId, quantity] of quantities.entries()) {
        const product = db.prepare(`
          SELECT id, name, sku, price, stock
          FROM products
          WHERE id = ? AND is_active = 1
        `).get(productId);

        if (!product) {
          const error = new Error(`Product ${productId} is unavailable.`);
          error.status = 400;
          throw error;
        }

        if (Number(product.stock) < quantity) {
          const error = new Error(`Insufficient stock for ${product.name}.`);
          error.status = 409;
          throw error;
        }

        const unitPrice = Number(product.price);

        if (!Number.isFinite(unitPrice) || unitPrice < 0) {
          const error = new Error(`Invalid price for ${product.name}.`);
          error.status = 500;
          throw error;
        }

        const lineTotal = Math.round(unitPrice * quantity * 100) / 100;
        subtotal += lineTotal;

        orderItems.push({
          product_id: product.id,
          product_name: product.name,
          sku: product.sku || null,
          quantity,
          unit_price: unitPrice,
          line_total: lineTotal,
          previous_stock: Number(product.stock)
        });
      }

      subtotal = Math.round(subtotal * 100) / 100;
      const totalAmount = Math.round((subtotal + deliveryFee) * 100) / 100;
      const orderNumber = `RS-${Date.now()}-${randomUUID().slice(0, 8).toUpperCase()}`;

      const orderResult = db.prepare(`
        INSERT INTO orders (
          order_number, customer_name, customer_email, customer_phone,
          delivery_address, delivery_city, delivery_state, delivery_pincode,
          subtotal, delivery_fee, discount, total_amount,
          payment_method, payment_status, status, customer_notes, customer_id
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
      `).run(
        orderNumber,
        customer.name,
        customer.email,
        customer.phone || "",
        clean(delivery_address),
        clean(delivery_city),
        clean(delivery_state),
        clean(delivery_pincode),
        subtotal,
        deliveryFee,
        0,
        totalAmount,
        "COD",
        "Pending",
        "Placed",
        clean(customer_notes),
        customer.id
      );

      const orderId = Number(orderResult.lastInsertRowid);

      const insertItem = db.prepare(`
        INSERT INTO order_items (
          order_id, product_id, product_name, sku,
          quantity, unit_price, line_total
        ) VALUES (?, ?, ?, ?, ?, ?, ?)
      `);

      const reduceStock = db.prepare(`
        UPDATE products
        SET stock = stock - ?
        WHERE id = ? AND is_active = 1 AND stock >= ?
      `);

      const insertInventoryLog = db.prepare(`
        INSERT INTO inventory_logs (
          product_id, change_amount, previous_stock, new_stock, reason
        ) VALUES (?, ?, ?, ?, ?)
      `);

      for (const item of orderItems) {
        const update = reduceStock.run(
          item.quantity,
          item.product_id,
          item.quantity
        );

        if (update.changes !== 1) {
          const error = new Error(
            `Stock changed for ${item.product_name}. Please try again.`
          );
          error.status = 409;
          throw error;
        }

        insertItem.run(
          orderId,
          item.product_id,
          item.product_name,
          item.sku,
          item.quantity,
          item.unit_price,
          item.line_total
        );

        insertInventoryLog.run(
          item.product_id,
          -item.quantity,
          item.previous_stock,
          item.previous_stock - item.quantity,
          `Customer order ${orderNumber}`
        );
      }

      db.prepare(`
        INSERT INTO order_events (
          order_id, old_status, new_status, note, changed_by
        ) VALUES (?, ?, ?, ?, ?)
      `).run(
        orderId,
        null,
        "Placed",
        "Order placed by customer",
        null
      );

      return {
        id: orderId,
        order_number: orderNumber,
        subtotal,
        delivery_fee: deliveryFee,
        discount: 0,
        total_amount: totalAmount,
        payment_method: "COD",
        payment_status: "Pending",
        status: "Placed",
        items: orderItems.map(({ previous_stock, ...item }) => item)
      };
    });

    const order = placeOrder();

    return res.status(201).json({
      success: true,
      message: "Order placed successfully.",
      order
    });
  } catch (error) {
    console.error("Customer order error:", error);

    return res.status(error.status || 500).json({
      success: false,
      message: error.status ? error.message : "Unable to place order."
    });
  }
});

router.patch("/:id/cancel", (req, res) => {
  try {
    const orderId = Number(req.params.id);
    const reason = clean(req.body?.reason);

    if (!Number.isSafeInteger(orderId) || orderId <= 0) {
      return res.status(400).json({
        success: false,
        message: "Invalid order ID."
      });
    }

    if (!reason || reason.length > 500) {
      return res.status(400).json({
        success: false,
        message: "A cancellation reason (maximum 500 characters) is required."
      });
    }

    const cancelOrder = db.transaction(() => {
      const order = db.prepare(`
        SELECT id, order_number, status
        FROM orders
        WHERE id = ? AND customer_id = ?
      `).get(orderId, req.customer.id);

      if (!order) {
        return { error: "NOT_FOUND" };
      }

      if (!["Placed", "Confirmed"].includes(order.status)) {
        return {
          error: "CANCELLATION_NOT_ALLOWED",
          current: order.status
        };
      }

      const items = db.prepare(`
        SELECT product_id, product_name, quantity
        FROM order_items
        WHERE order_id = ?
      `).all(orderId);

      const restoreStock = db.prepare(`
        UPDATE products
        SET stock = stock + ?
        WHERE id = ?
      `);

      const insertLog = db.prepare(`
        INSERT INTO inventory_logs (
          product_id, change_amount, previous_stock, new_stock, reason
        ) VALUES (?, ?, ?, ?, ?)
      `);

      for (const item of items) {
        const product = db.prepare(`
          SELECT stock FROM products WHERE id = ?
        `).get(item.product_id);

        if (!product) {
          throw new Error(`Product ${item.product_id} not found.`);
        }

        const previousStock = Number(product.stock);
        const quantity = Number(item.quantity);
        const newStock = previousStock + quantity;

        restoreStock.run(quantity, item.product_id);

        insertLog.run(
          item.product_id,
          quantity,
          previousStock,
          newStock,
          `Customer cancelled order ${order.order_number}`
        );
      }

      db.prepare(`
        UPDATE orders
        SET status = 'Cancelled',
            cancellation_reason = ?,
            updated_at = CURRENT_TIMESTAMP
        WHERE id = ? AND customer_id = ?
      `).run(reason, orderId, req.customer.id);

      db.prepare(`
        INSERT INTO order_events (
          order_id, old_status, new_status, note, changed_by
        ) VALUES (?, ?, ?, ?, NULL)
      `).run(
        orderId,
        order.status,
        "Cancelled",
        reason
      );

      return {
        success: true,
        old_status: order.status,
        new_status: "Cancelled"
      };
    });

    const result = cancelOrder();

    if (result.error === "NOT_FOUND") {
      return res.status(404).json({
        success: false,
        message: "Order not found."
      });
    }

    if (result.error === "CANCELLATION_NOT_ALLOWED") {
      return res.status(409).json({
        success: false,
        message: `Order cannot be cancelled from ${result.current} status.`
      });
    }

    return res.json({
      success: true,
      message: "Order cancelled and inventory restored.",
      data: result
    });
  } catch (error) {
    console.error("Customer order cancellation error:", error);

    return res.status(500).json({
      success: false,
      message: "Unable to cancel order."
    });
  }
});

export default router;

import { Router } from "express";
import bcrypt from "bcryptjs";
import jwt from "jsonwebtoken";

import db from "../database.js";
import { requireCustomer } from "../middleware/customer-auth.js";

const router = Router();

function createToken(customerId) {
  const secret = process.env.JWT_SECRET;

  if (!secret || secret.length < 32) {
    throw new Error("JWT_SECRET must contain at least 32 characters");
  }

  return jwt.sign(
    { role: "customer" },
    secret,
    {
      subject: String(customerId),
      issuer: "riya-quick-commerce",
      expiresIn: "7d",
      algorithm: "HS256"
    }
  );
}

function publicCustomer(customer) {
  return {
    id: customer.id,
    name: customer.name,
    email: customer.email,
    phone: customer.phone
  };
}

// CUSTOMER REGISTRATION
router.post("/register", (req, res) => {
  try {
    const name = String(req.body?.name || "").trim();
    const email = String(req.body?.email || "").trim().toLowerCase();
    const phone = String(req.body?.phone || "").trim();
    const password = String(req.body?.password || "");

    if (name.length < 2 || name.length > 100) {
      return res.status(400).json({
        success: false,
        message: "Name must contain 2 to 100 characters"
      });
    }

    if (
      email.length > 254 ||
      !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)
    ) {
      return res.status(400).json({
        success: false,
        message: "Enter a valid email address"
      });
    }

    if (!/^\+?[0-9\s-]{10,18}$/.test(phone)) {
      return res.status(400).json({
        success: false,
        message: "Enter a valid phone number"
      });
    }

    if (
      password.length < 8 ||
      Buffer.byteLength(password, "utf8") > 72
    ) {
      return res.status(400).json({
        success: false,
        message: "Password must be at least 8 characters and at most 72 bytes"
      });
    }

    const passwordHash = bcrypt.hashSync(password, 12);

    const result = db.prepare(`
      INSERT INTO customers (name, email, phone, password_hash)
      VALUES (?, ?, ?, ?)
    `).run(name, email, phone, passwordHash);

    const customer = db.prepare(`
      SELECT id, name, email, phone
      FROM customers
      WHERE id = ?
    `).get(result.lastInsertRowid);

    const token = createToken(customer.id);

    return res.status(201).json({
      success: true,
      message: "Customer account created successfully",
      data: {
        token,
        customer: publicCustomer(customer)
      }
    });
  } catch (error) {
    if (error.code === "SQLITE_CONSTRAINT_UNIQUE") {
      return res.status(409).json({
        success: false,
        message: "An account with this email already exists"
      });
    }

    console.error("Customer registration error:", error.message);

    return res.status(500).json({
      success: false,
      message: "Unable to create customer account"
    });
  }
});

router.post("/login", (req, res) => {
  try {
    const email = String(req.body?.email || "").trim().toLowerCase();
    const password = String(req.body?.password || "");

    if (!email || !password || password.length > 72) {
      return res.status(400).json({
        success: false,
        message: "Email and password are required"
      });
    }

    const customer = db.prepare(`
      SELECT id, name, email, phone, password_hash
      FROM customers
      WHERE email = ? AND is_active = 1
    `).get(email);

    if (
      !customer ||
      !bcrypt.compareSync(password, customer.password_hash)
    ) {
      return res.status(401).json({
        success: false,
        message: "Invalid email or password"
      });
    }

    const token = createToken(customer.id);

    return res.json({
      success: true,
      message: "Login successful",
      data: {
        token,
        customer: publicCustomer(customer)
      }
    });
  } catch (error) {
    console.error("Customer login error:", error.message);

    return res.status(500).json({
      success: false,
      message: "Unable to login"
    });
  }
});

import { Router } from "express";
import bcrypt from "bcryptjs";
import jwt from "jsonwebtoken";
import db from "../database.js";

const router = Router();

  const secret = process.env.JWT_SECRET;

  if (!secret || secret.length < 32) {
    throw new Error("JWT_SECRET is not configured correctly.");
  }

  return jwt.sign(
    { role: "rider" },
    secret,

      issuer: "riya-quick-commerce",
      expiresIn: "8h",

router.post("/login", async (req, res) => {
  try {
    const identifier = String(req.body?.identifier || "")
      .trim()
      .toLowerCase();

    const password = String(req.body?.password || "");

    if (!identifier || !password || password.length > 72) {
      return res.status(400).json({
        success: false,
        message: "Valid phone/email and password are required"
      });
    }

    const rider = db.prepare(`
      SELECT id, name, phone, email, password_hash, is_active
      FROM riders
      WHERE (LOWER(phone) = ? OR LOWER(email) = ?)
      LIMIT 1
    `).get(identifier, identifier);

    if (
      !rider ||
      rider.is_active !== 1 ||
      !(await bcrypt.compare(password, rider.password_hash))
    ) {
      return res.status(401).json({
        success: false,
        message: "Invalid login credentials"
      });
    }

    const token = createToken(rider);

    return res.json({
      success: true,
      message: "Rider login successful",
      token,
      rider: {
        id: rider.id,
        name: rider.name,
        phone: rider.phone,
        email: rider.email
      }
    });
  } catch (error) {
    console.error("Rider login error:", error);

    return res.status(500).json({
      success: false,
      message: "Unable to login"
    });
  }
});

  });

import jwt from "jsonwebtoken";
import db from "../database.js";

export function requireRider(req, res, next) {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader?.startsWith("Bearer ")) {
      return res.status(401).json({
        success: false,
        message: "Rider authentication required"
      });
    }

    const token = authHeader.slice(7).trim();
    const secret = process.env.JWT_SECRET;

    if (!secret || secret.length < 32) {
      return res.status(500).json({
        success: false,
        message: "Server authentication is not configured"
      });
    }

    const decoded = jwt.verify(token, secret, {
      issuer: "riya-quick-commerce",
      algorithms: ["HS256"]
    });

    const riderId = Number(decoded.sub);

    if (
      !Number.isSafeInteger(riderId) ||
      riderId < 1 ||
      decoded.role !== "rider"
    ) {
      return res.status(401).json({
        success: false,
        message: "Invalid rider token"
      });
    }

    const rider = db.prepare(`
      SELECT id, name, phone, email, is_active
      FROM riders
      WHERE id = ? AND is_active = 1
    `).get(riderId);

    if (!rider) {
      return res.status(401).json({
        success: false,
        message: "Rider account is inactive or unavailable"
      });
    }

    req.rider = rider;
    next();
  } catch (error) {
    if (
      error.name === "TokenExpiredError" ||
      error.name === "JsonWebTokenError"
    ) {
      return res.status(401).json({
        success: false,
        message: "Invalid or expired rider token"
      });
    }

    console.error("Rider authentication error:", error);

    return res.status(500).json({
      success: false,
      message: "Rider authentication failed"
    });
  }
}

import jwt from "jsonwebtoken";
import db from "../database.js";

export function requireAdmin(req, res, next) {
  try {
    // 1. Check authorization header
    const authHeader = req.headers.authorization;

    if (!authHeader || !authHeader.startsWith("Bearer ")) {
      return res.status(401).json({
        success: false,
        message: "Authentication required"
      });
    }

    // 2. Extract token
    const token = authHeader.slice(7).trim();

    if (!token) {
      return res.status(401).json({
        success: false,
        message: "Authentication token is missing"
      });
    }

    // 3. Check JWT secret
    const secret = process.env.JWT_SECRET;

    if (!secret || secret.length < 32) {
      return res.status(500).json({
        success: false,
        message: "Server authentication is not configured"
      });
    }

    // 4. Verify token
    const decoded = jwt.verify(token, secret, {
      issuer: "riya-quick-commerce",
      algorithms: ["HS256"]
    });

    // 5. Validate admin ID from JWT subject
    const adminId = Number(decoded.sub);

    if (
      !Number.isSafeInteger(adminId) ||
      adminId < 1 ||
      decoded.role !== "admin"
    ) {
      return res.status(401).json({
        success: false,
        message: "Invalid authentication token"
      });
    }

    // 6. Check admin account in database
    const admin = db.prepare(`
      SELECT
        id,
        email,
        role,
        is_active
      FROM admin_users
      WHERE id = ?
        AND is_active = 1
    `).get(adminId);

    if (!admin) {
      return res.status(401).json({
        success: false,
        message: "Admin account is inactive or unavailable"
      });
    }

    // 7. Attach authenticated admin
    req.admin = {
      id: admin.id,
      email: admin.email,
      role: admin.role
    };

    return next();

  } catch (error) {
    if (error.name === "TokenExpiredError") {
      return res.status(401).json({
        success: false,
        message: "Authentication token has expired"
      });
    }

    if (error.name === "JsonWebTokenError") {
      return res.status(401).json({
        success: false,
        message: "Invalid authentication token"
      });
    }

    console.error("Admin authentication error:", error.message);

    return res.status(500).json({
      success: false,
      message: "Authentication failed"
    });
  }
}

  try {
    const riders = db.prepare(`
      SELECT
        r.id,
        r.name,
        r.phone,
        r.email,
        r.is_active,
        r.created_at,
        r.updated_at,
        COUNT(o.id) AS assigned_orders,
        SUM(CASE WHEN o.status = 'Out for Delivery' THEN 1 ELSE 0 END)
          AS active_deliveries
      FROM riders r
      LEFT JOIN orders o ON o.assigned_rider_id = r.id
      GROUP BY r.id
      ORDER BY r.created_at DESC, r.id DESC
    `).all();

    return res.json({
      success: true,
      data: riders
    });
  } catch (error) {
    console.error("Admin riders list error:", error);
    return res.status(500).json({
      success: false,
      message: "Unable to load riders"
    });
  }
});

// POST /api/admin/riders

      return res.status(500).json({
        success: false,
        message: "Invalid delivery fee configuration."
      });

      const customer = db.prepare(`

      !(await bcrypt.compare(password, rider.password_hash))
    ) {
      return res.status(401).json({
        success: false,
        message: "Invalid login credentials"
      });
    }

    const token = createToken(rider);

    return res.json({
      success: true,
      message: "Rider login successful",
      token,
      rider: {
        id: rider.id,
        name: rider.name,
        phone: rider.phone,
        email: rider.email
      }
    });
  } catch (error) {
    console.error("Rider login error:", error);

    return res.status(500).json({
      success: false,
      message: "Unable to login"
    });
  }
});

import jwt from "jsonwebtoken";
import db from "../database.js";

export function requireRider(req, res, next) {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader?.startsWith("Bearer ")) {
      return res.status(401).json({
        success: false,
        message: "Rider authentication required"
      });
    }

    const token = authHeader.slice(7).trim();
    const secret = process.env.JWT_SECRET;

    if (!secret || secret.length < 32) {
      return res.status(500).json({
        success: false,
        message: "Server authentication is not configured"
      });
    }

    const decoded = jwt.verify(token, secret, {
      issuer: "riya-quick-commerce",
      algorithms: ["HS256"]
    });

    const riderId = Number(decoded.sub);

    if (
      !Number.isSafeInteger(riderId) ||
      riderId < 1 ||
      decoded.role !== "rider"
    ) {
      return res.status(401).json({
        success: false,
        message: "Invalid rider token"
      });
    }

    const rider = db.prepare(`
      SELECT id, name, phone, email, is_active
      FROM riders
      WHERE id = ? AND is_active = 1
    `).get(riderId);

    if (!rider) {
      return res.status(401).json({
        success: false,
        message: "Rider account is inactive or unavailable"
      });
    }

    req.rider = rider;
    next();
  } catch (error) {
    if (
      error.name === "TokenExpiredError" ||
      error.name === "JsonWebTokenError"
    ) {
      return res.status(401).json({
        success: false,
        message: "Invalid or expired rider token"
      });
    }

    console.error("Rider authentication error:", error);

    return res.status(500).json({
      success: false,
      message: "Rider authentication failed"
    });
  }
}

import jwt from "jsonwebtoken";
import db from "../database.js";

export function requireAdmin(req, res, next) {
  try {
    // 1. Check authorization header
    const authHeader = req.headers.authorization;

    if (!authHeader || !authHeader.startsWith("Bearer ")) {
      return res.status(401).json({
        success: false,
        message: "Authentication required"
      });
    }

    // 2. Extract token
    const token = authHeader.slice(7).trim();

    if (!token) {
      return res.status(401).json({
        success: false,
        message: "Authentication token is missing"
      });
    }

    // 3. Check JWT secret
    const secret = process.env.JWT_SECRET;

    if (!secret || secret.length < 32) {
      return res.status(500).json({
        success: false,
        message: "Server authentication is not configured"
      });
    }

    // 4. Verify token
    const decoded = jwt.verify(token, secret, {
      issuer: "riya-quick-commerce",
      algorithms: ["HS256"]
    });

    // 5. Validate admin ID from JWT subject
    const adminId = Number(decoded.sub);

    if (
      !Number.isSafeInteger(adminId) ||
      adminId < 1 ||
      decoded.role !== "admin"
    ) {
      return res.status(401).json({
        success: false,
        message: "Invalid authentication token"
      });
    }

    // 6. Check admin account in database
    const admin = db.prepare(`
      SELECT
        id,
        email,
        role,
        is_active
      FROM admin_users
      WHERE id = ?
        AND is_active = 1
    `).get(adminId);

    if (!admin) {
      return res.status(401).json({
        success: false,
        message: "Admin account is inactive or unavailable"
      });
    }

    // 7. Attach authenticated admin
    req.admin = {
      id: admin.id,
      email: admin.email,
      role: admin.role
    };

    return next();

  } catch (error) {
    if (error.name === "TokenExpiredError") {
      return res.status(401).json({
        success: false,
        message: "Authentication token has expired"
      });
    }

    if (error.name === "JsonWebTokenError") {
      return res.status(401).json({
        success: false,
        message: "Invalid authentication token"
      });
    }

    console.error("Admin authentication error:", error.message);

    return res.status(500).json({
      success: false,
      message: "Authentication failed"
    });
  }
}
