function normalizeCategory(category: string) {
  const normalized = category.trim().toLowerCase()

  if (normalized === 'home & living' || normalized === 'home and living') {
    return 'Home'
  }

  if (normalized === 'personal care') {
    return 'Personal Care'
  }

  return category
}

function mapApiProduct(item: ApiProduct): Product {
  const category = normalizeCategory(item.category || 'Other')

  return {
    id: item.id,
    name: item.name,
    category,
    description: item.description || 'Quality product from Riya Quick Commerce',
    price: Number(item.price) || 0,
    mrp: Number(item.originalPrice ?? item.price) || 0,
    unit: '1 piece',
    image: item.imageUrl || fallbackImages[category] ||
      'https://images.unsplash.com/photo-1542291026-7eec264c27ff?w=600',
    rating: 0,
  }
}

const formatPrice = (price: number) =>
  `₹${price.toLocaleString('en-IN')}`

        throw new Error('Invalid product data received from server.')
      }

      setProducts(response.data.map(mapApiProduct))
    } catch (error) {
      console.error('Product loading error:', error)
      setApiError(
        error instanceof Error
          ? error.message
          : 'Unable to connect to the product server.'
      )
    } finally {
      setLoading(false)
    }

    void loadProducts()

    return products.filter((product) => {

        activeCategory === 'All' || product.category === activeCategory

        !query ||
        product.name.toLowerCase().includes(query) ||
        product.category.toLowerCase().includes(query) ||
        product.description.toLowerCase().includes(query)

      return matchesCategory && matchesSearch

    setCart((previous) => {

      return next

                          event.currentTarget.src =
                            'https://images.unsplash.com/photo-1542291026-7eec264c27ff?w=600'

              <div className="empty-cart">
                <div className="empty-cart-icon">
                  <ShoppingCart size={48} />
                </div>
                <h3>Your cart is empty</h3>
                <p>Add products to start shopping.</p>
                <button
                  className="primary-button"
                  onClick={() => setCartOpen(false)}
                >
                  Explore products
                </button>
              </div>
            ) : (
              <>
                <div className="cart-items">
                  {cartItems.map((product) => (
                    <div className="cart-item" key={product.id}>
                      <img src={product.image} alt={product.name} />

                      <div className="cart-item-info">
                        <strong>{product.name}</strong>
                        <small>{product.unit}</small>
                        <b>
                          {formatPrice(product.price * cart[product.id])}
                        </b>
                      </div>

                      <div className="quantity-control">
                        <button onClick={() => updateCart(product.id, -1)}>
                          <Minus size={13} />
                        </button>
                        <strong>{cart[product.id]}</strong>
                        <button onClick={() => updateCart(product.id, 1)}>
                          <Plus size={13} />
                        </button>
                      </div>
                    </div>
                  ))}
                </div>

                <div className="cart-summary">
                  <div>
                    <span>Subtotal</span>
                    <strong>{formatPrice(subtotal)}</strong>
                  </div>

                  <div>
                    <span>Product savings</span>
                    <strong className="savings-text">
                      -{formatPrice(totalSavings)}
                    </strong>
                  </div>

                  <div>
                    <span>Delivery charge</span>
                    <strong>Calculated at checkout</strong>
                  </div>

                  <div className="cart-total">
                    <span>Items total</span>
                    <strong>{formatPrice(subtotal)}</strong>
                  </div>

                  <button
                    className="checkout-button"
                    onClick={() =>
                      alert('Checkout will be implemented in the next development phase.')
                    }
                  >
                    Continue to checkout <ArrowRight size={18} />
                  </button>

                  <small>
                    Final payable amount will be calculated at checkout.
                  </small>
                </div>
              </>
            )}
          </aside>
        </div>
      )}
    </div>
  )
}

const API_URL = import.meta.env.VITE_API_URL || 'http://localhost:5000/api'

export type ApiProduct = {
  id: number
  name: string
  slug: string
  description: string
  price: number
  originalPrice: number | null
  stock: number
  imageUrl: string
  sku: string | null
  category: string
  categorySlug: string
}

export type ApiCategory = {
  id: number
  name: string
  slug: string
  description: string
  product_count: number
}

type ApiResponse<T> = {
  success: boolean
  data: T
  message?: string
  total?: number
}

async function request<T>(endpoint: string): Promise<ApiResponse<T>> {
  const response = await fetch(`${API_URL}${endpoint}`)

  if (!response.ok) {
    throw new Error(`API request failed: ${response.status}`)
  }

  return response.json() as Promise<ApiResponse<T>>
}

export function getProducts(search = '', category = '') {
  const params = new URLSearchParams()

  if (search.trim()) params.set('search', search.trim())
  if (category && category !== 'All') {
    params.set('category', category.toLowerCase().replaceAll(' ', '-').replaceAll('&-', ''))
  }

  const query = params.toString()

  return request<ApiProduct[]>(`/products${query ? `?${query}` : ''}`)
}

export function getCategories() {
  return request<ApiCategory[]>('/categories')
}

export function checkApiHealth() {
  return request<{
    message: string
    database: string
    timestamp: string
  }>('/health')
}

async function apiRequest<T>(
  endpoint: string,
  token?: string,
  options: RequestInit = {},
): Promise<T> {
  const headers = new Headers(options.headers);

  if (options.body && !headers.has('Content-Type')) {
    headers.set('Content-Type', 'application/json');
  }

  if (token) {
    headers.set('Authorization', `Bearer ${token}`);
  }

  const response = await fetch(`${API_URL}${endpoint}`, {
    ...options,
    headers,
  });

  const result = await response.json().catch(() => ({}));

  if (!response.ok) {
    throw new Error(
      result.message || result.error || `Request failed (${response.status})`,
    );
  }

  return result as T;
}

function formatCurrency(value: number | null | undefined) {
  return new Intl.NumberFormat('en-IN', {
    style: 'currency',
    currency: 'INR',
    maximumFractionDigits: 0,
  }).format(Number(value || 0));
}

function formatDate(value: string | undefined) {
  if (!value) return 'â€”';

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) return value;

  return date.toLocaleString('en-IN', {
    dateStyle: 'medium',
    timeStyle: 'short',
  });
}

function isActiveProduct(product: Product) {
  return product.isActive === true || product.isActive === 1;
}

      return;
    }

      setError('Stock must be a whole number greater than or equal to zero.');
      return;
    }

    const payload = {
      name: productForm.name.trim(),
      description: productForm.description.trim(),
      categoryId: Number(productForm.categoryId),
      price,
      originalPrice,
      stock,
      sku: productForm.sku.trim(),
      imageUrl: productForm.imageUrl.trim(),

      if (editingProduct) {

          method: 'POST',

function StockBadge({ stock }: { stock: number }) {
  if (stock === 0) {

  try {
    const search = String(req.query.search || "").trim().slice(0, 100);
    const category = String(req.query.category || "").trim().slice(0, 80);

    const page = Math.max(
      1,
      Number.parseInt(req.query.page, 10) || 1
    );

    const limit = Math.min(
      50,
      Math.max(1, Number.parseInt(req.query.limit, 10) || 20)
    );

    const conditions = ["p.is_active = 1"];
    const params = [];

    if (search) {
      conditions.push("(p.name LIKE ? OR p.description LIKE ?)");
      params.push(`%${search}%`, `%${search}%`);
    }

    if (category) {
      conditions.push("c.slug = ?");
      params.push(category.toLowerCase());
    }

    const where = conditions.join(" AND ");

    const countResult = db.prepare(`
      SELECT COUNT(*) AS total
      FROM products p
      LEFT JOIN categories c ON c.id = p.category_id
      WHERE ${where}
    `).get(...params);

    const products = db.prepare(`
      SELECT
        p.id,
        p.name,
        p.slug,
        p.description,
        p.price,
        p.original_price AS originalPrice,
        p.stock,
        p.image_url AS imageUrl,
        p.sku,
        c.name AS category,
        c.slug AS categorySlug
      FROM products p
      LEFT JOIN categories c ON c.id = p.category_id
      WHERE ${where}
      ORDER BY p.created_at DESC, p.id DESC
      LIMIT ? OFFSET ?
    `).all(...params, limit, (page - 1) * limit);

    res.json({
      success: true,
      count: products.length,
      total: countResult.total,
      page,
      limit,
      data: products
    });

  } catch {
    res.status(500).json({
      success: false,
      message: "Unable to fetch products"
    });
  }
});
