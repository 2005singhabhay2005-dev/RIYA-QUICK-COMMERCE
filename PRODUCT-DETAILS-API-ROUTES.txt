function slugify(value) {
  return String(value)
    .normalize("NFKD")
    .toLowerCase()
    .trim()
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-|-$/g, "");
}

function audit(adminId, action, type, id, details = "") {
  db.prepare(`
    INSERT INTO admin_audit_logs
    (admin_id, action, entity_type, entity_id, details)
    VALUES (?, ?, ?, ?, ?)
  `).run(adminId, action, type, id ?? null, details);
}

function validText(value, max, field) {
  if (
    typeof value !== "string" ||
    !value.trim() ||
    value.trim().length > max
  ) {
    throw new Error(`${field} is required and must be under ${max} characters`);
  }

  return value.trim();
}

function validMoney(value, field, nullable = false) {
  if (
    nullable &&
    (value === null || value === "" || value === undefined)
  ) {
    return null;
  }

  const number = Number(value);

  if (
    !Number.isFinite(number) ||
    number < 0 ||
    number > 100000000
  ) {
    throw new Error(`${field} must be a valid non-negative amount`);
  }

  return number;
}

function validStock(value) {
  const number = Number(value);

  if (
    !Number.isInteger(number) ||
    number < 0 ||
    number > 100000000
  ) {
    throw new Error("Stock must be a non-negative whole number");
  }

  return number;
}

function sendError(res, error) {
  const message = error?.message || "Request failed";

  if (message.includes("UNIQUE constraint failed")) {
    return res.status(409).json({
      success: false,
      message: "A record with this name, slug, or SKU already exists"
    });
  }

  return res.status(400).json({
    success: false,
    message
  });
}

router.post("/login", (req, res) => {
  const email = String(req.body?.email || "").trim().toLowerCase();
  const password = String(req.body?.password || "");

  if (!email || !password || password.length > 200) {
    return res.status(400).json({
      success: false,
      message: "Email and password are required"
    });
  }

  const admin = db.prepare(`
    SELECT *
    FROM admin_users
    WHERE email = ? AND is_active = 1
  `).get(email);

  if (!admin || !bcrypt.compareSync(password, admin.password_hash)) {
    return res.status(401).json({
      success: false,
      message: "Invalid email or password"
    });
  }

  const secret = process.env.JWT_SECRET;

  if (!secret || secret.length < 32) {
    return res.status(500).json({
      success: false,
      message: "JWT_SECRET must contain at least 32 characters"
    });
  }

  const token = jwt.sign(
    { role: admin.role },
    secret,
    {
      subject: String(admin.id),
      issuer: "riya-quick-commerce",
      expiresIn: "8h"
    }
  );

  res.json({
    success: true,
    data: {
      token,
      admin: {
        id: admin.id,
        email: admin.email,
        role: admin.role
      }
    }
  });
});

router.post("/products", (req, res) => {
  try {
    const body = req.body || {};

    const name = validText(body.name, 150, "Product name");

    const description =
      typeof body.description === "string"
        ? body.description.trim().slice(0, 2000)
        : "";

    const categoryId = Number(body.categoryId);

    const categoryExists = db.prepare(`
      SELECT id FROM categories WHERE id = ?
    `).get(categoryId);

    if (!Number.isInteger(categoryId) || !categoryExists) {
      throw new Error("Select a valid category");
    }

    const price = validMoney(body.price, "Price");
    const originalPrice = validMoney(
      body.originalPrice,
      "MRP",
      true
    );

    const stock = validStock(body.stock ?? 0);

    const sku = body.sku
      ? validText(body.sku, 80, "SKU")
      : null;

    const imageUrl =
      typeof body.imageUrl === "string"
        ? body.imageUrl.trim().slice(0, 1000)
        : "";

    if (imageUrl && !/^https?:\/\/\S+$/i.test(imageUrl)) {
      throw new Error("Image URL must start with http:// or https://");
    }

    if (originalPrice !== null && originalPrice < price) {
      throw new Error("MRP cannot be lower than selling price");
    }

    const slug = slugify(name);

    if (!slug) {
      throw new Error("Invalid product name");
    }

    const transaction = db.transaction(() => {
      const result = db.prepare(`
        INSERT INTO products
        (
          name,
          slug,
          description,
          category_id,
          price,
          original_price,
          stock,
          image_url,
          sku
        )
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
      `).run(
        name,
        slug,
        description,
        categoryId,
        price,
        originalPrice,
        stock,
        imageUrl,
        sku
      );

      if (stock > 0) {
        db.prepare(`
          INSERT INTO inventory_logs
          (
            product_id,
            change_amount,
            previous_stock,
            new_stock,
            reason
          )
          VALUES (?, ?, 0, ?, 'Initial stock')
        `).run(result.lastInsertRowid, stock, stock);
      }

      audit(
        req.admin.id,
        "create",
        "product",
        result.lastInsertRowid,
        name
      );

      return result.lastInsertRowid;
    });

    const id = transaction();

    const product = db.prepare(`
      ${productSelect}
      WHERE p.id = ?
    `).get(id);

    res.status(201).json({
      success: true,
      data: product
    });

  } catch (error) {
    sendError(res, error);
  }
});

        FROM products
        WHERE id = ?
      `).get(id);

      if (!product) {
        throw new Error("Product not found");
      }

      const change = stock - product.stock;

      db.prepare(`
        UPDATE products
        SET stock = ?, updated_at = CURRENT_TIMESTAMP
        WHERE id = ?
      `).run(stock, id);

      db.prepare(`
        INSERT INTO inventory_logs
        (
          product_id,
          change_amount,
          previous_stock,
          new_stock,
          reason
        )
        VALUES (?, ?, ?, ?, ?)
      `).run(
        id,
        change,
        product.stock,
        stock,
        reason
      );

      audit(
        req.admin.id,
        "stock_update",
        "product",
        id,
        `${product.stock} -> ${stock}: ${reason}`
      );

      return {
        previousStock: product.stock,
        newStock: stock,
        changeAmount: change
      };
    });

    res.json({
      success: true,
      data: transaction()
    });

  } catch (error) {
    const status = error.message === "Product not found" ? 404 : 400;

    res.status(status).json({
      success: false,
      message: error.message
    });
  }
});

  try {
    const {
      search = "",

      page = "1",
      limit = "20"
    } = req.query;

    const currentPage = Math.max(1, Number.parseInt(page, 10) || 1);

    );

    const conditions = [];
    const params = [];

    if (status && STATUSES.includes(status)) {
      conditions.push("o.status = ?");

      conditions.push(`(

        OR o.customer_phone LIKE ?
        OR o.customer_email LIKE ?
      )`);
      params.push(term, term, term, term);
    }

    const where = conditions.length
      ? `WHERE ${conditions.join(" AND ")}`
      : "";

    const total = db.prepare(`
      SELECT COUNT(*) AS count

      ${where}
      GROUP BY o.id
      ORDER BY o.created_at DESC, o.id DESC
      LIMIT ? OFFSET ?
    `).all(...params, pageSize, (currentPage - 1) * pageSize);

      success: true,

        reason,
        req.admin.id
      );

      return {
        success: true,
        old_status: oldStatus,
        new_status: status
      };
    });

    const result = updateOrder();

    if (result.error === "NOT_FOUND") {
      return res.status(404).json({
        success: false,
        message: "Order not found"
      });
    }

    if (result.error === "CANCELLATION_REASON_REQUIRED") {
      return res.status(400).json({
        success: false,
        message: "Cancellation reason is required"
      });
    }

    if (result.error === "INVALID_TRANSITION") {
      return res.status(409).json({
        success: false,
        message: `Cannot change order from ${result.current} to ${status}`
      });
    }

    res.json({
      success: true,
      message: "Order status updated",
      data: result
    });
  } catch (error) {
    console.error("Order status update error:", error);
    res.status(500).json({
      success: false,
      message: "Unable to update order status"
    });
  }
});

export default router;
